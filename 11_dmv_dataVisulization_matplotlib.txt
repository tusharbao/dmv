
import pandas as pd
import matplotlib.pyplot as plt

# Load dataset
df = pd.read_csv("city_day.csv")

# Explore dataset
print("First 5 Records:")
print(df.head())

print("\nDataset Shape:")
print(df.shape)

print("\nColumn Names:")
print(df.columns.tolist())

print("\nDataset Information:")
df.info()

print("\nMissing Values:")
print(df.isnull().sum())

# Convert Date format
df["Date"] = pd.to_datetime(df["Date"])

# Sort by date
df = df.sort_values("Date")

# Remove missing AQI values
df = df.dropna(subset=["AQI"])

# Summary statistics
print("\nSummary Statistics:")
print(df.describe())

# 1. AQI Trend
plt.figure(figsize=(10,5))
plt.plot(df["Date"], df["AQI"], marker="o")

plt.title("AQI Trend Over Time")
plt.xlabel("Date")
plt.ylabel("AQI")
plt.xticks(rotation=45)
plt.grid(True)
plt.tight_layout()
plt.show()

# 2. Pollutant Levels
plt.figure(figsize=(10,5))

plt.plot(df["Date"], df["PM2.5"], label="PM2.5")
plt.plot(df["Date"], df["PM10"], label="PM10")
plt.plot(df["Date"], df["CO"], label="CO")

plt.title("Pollutant Levels Over Time")
plt.xlabel("Date")
plt.ylabel("Pollutant Level")
plt.legend()
plt.xticks(rotation=45)
plt.grid(True)
plt.tight_layout()
plt.show()

# 3. AQI Bar Plot
plt.figure(figsize=(10,5))
plt.bar(df["Date"], df["AQI"])

plt.title("AQI Comparison by Date")
plt.xlabel("Date")
plt.ylabel("AQI")
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

# 4. AQI Box Plot
plt.figure(figsize=(7,5))
plt.boxplot(df["AQI"].dropna())

plt.title("AQI Distribution")
plt.ylabel("AQI")
plt.show()

# 5. PM2.5 vs AQI
plt.figure(figsize=(8,5))
data = df.dropna(subset=["PM2.5", "AQI"])
plt.scatter(data["PM2.5"], data["AQI"])

plt.title("PM2.5 vs AQI")
plt.xlabel("PM2.5")
plt.ylabel("AQI")
plt.grid(True)
plt.show()

# 6. PM10 vs AQI
plt.figure(figsize=(8,5))
data = df.dropna(subset=["PM10", "AQI"])
plt.scatter(data["PM10"], data["AQI"])

plt.title("PM10 vs AQI")
plt.xlabel("PM10")
plt.ylabel("AQI")
plt.grid(True)
plt.show()

# 7. CO vs AQI
plt.figure(figsize=(8,5))
data = df.dropna(subset=["CO", "AQI"])
plt.scatter(data["CO"], data["AQI"])

plt.title("CO vs AQI")
plt.xlabel("CO")
plt.ylabel("AQI")
plt.grid(True)
plt.show()

print("\nData Visualization Completed Successfully!")
